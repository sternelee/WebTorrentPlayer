# 下载功能实施计划

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 TorPlay 添加文件下载到磁盘功能，支持多任务队列管理、全局限速、暂停/恢复控制，覆盖 Desktop / Android / iOS 三个平台。

**Architecture:**
- **Backend**: 新增 `download.rs` 模块，封装 librqbit 的下载到磁盘逻辑。Desktop 直接写入 Downloads/TorPlay；Mobile 使用 cache 后导出。
- **Frontend**: Downloads Tab（与 Stream Tab 并列），全局限速控件，批量操作按钮。
- **IPC**: 新增 7 个命令，管理下载生命周期与队列。

**Tech Stack:** librqbit, axum, SolidJS, Tauri 2

pub fn list_download_files(
    info_hash: String,
    state: State<'_, Arc<AppState>>,
) -> Result<Vec<FileInfo>, String> {
    let handle = state.torrent(&info_hash).map_err(anyhow_to_string)?;
    let download_dir = state.download_dir.read().clone();

    handle
        .with_metadata(|metadata| {
            metadata
                .file_infos
                .iter()
                .enumerate()
                .map(|(index, file)| {
                    let downloaded = handle
                        .stats()
                        .file_progress
                        .get(index)
                        .copied()
                        .unwrap_or(0);
                    FileInfo {
                        index,
                        name: file.relative_filename.to_string_lossy().into_owned(),
                        size_bytes: file.len,
                        downloaded_bytes: downloaded,
                        path: Some(download_dir.join(&file.relative_filename).to_string_lossy().into_owned()),
                        is_video: is_video_path(&file.relative_filename),
                    }
                })
                .collect()
        })
        .map_err(anyhow_to_string)

- [ ] **Step 3: 实现 get_active_downloads 命令**

遍历所有 monitored torrents，转换为 DownloadTask:

```rust
#[tauri::command]
pub fn get_active_downloads(state: State<'_, Arc<AppState>>) -> Vec<DownloadTask> {
    let info_hashes: Vec<String> = state.monitored_torrents.read().iter().cloned().collect();
    let download_dir = state.download_dir.read().clone();

    info_hashes
        .into_iter()
        .filter_map(|info_hash| {
            let handle = state.torrent(&info_hash).ok()?;
            let stats = handle.stats();

            let files = handle.with_metadata(|metadata| {
                metadata.file_infos.iter().enumerate().map(|(index, file)| {
                    let downloaded = stats.file_progress.get(index).copied().unwrap_or(0);
                    FileInfo {
                        index,
                        name: file.relative_filename.to_string_lossy().into_owned(),
                        size_bytes: file.len,
                        downloaded_bytes: downloaded,
                        path: Some(download_dir.join(&file.relative_filename).to_string_lossy().into_owned()),
                        is_video: is_video_path(&file.relative_filename),
                    }
                }).collect()
            }).ok()?;

            let name = files.iter().find(|f| f.is_video).map(|f| f.name.clone())
                .unwrap_or_else(|| files.first().map(|f| f.name.clone()).unwrap_or_default());

            Some(DownloadTask {
                info_hash: info_hash.clone(),
                name,
                progress_percent: if stats.total_bytes == 0 {
                    0.0
                } else {
                    stats.progress_bytes as f64 / stats.total_bytes as f64 * 100.0
                },
                download_speed_bps: stats.live.as_ref().map(|l| (l.download_speed.mbps * 1024.0 * 1024.0) as u64).unwrap_or(0),
                upload_speed_bps: stats.live.as_ref().map(|l| (l.upload_speed.mbps * 1024.0 * 1024.0) as u64).unwrap_or(0),
                peers_connected: stats.live.as_ref().map(|l| l.snapshot.peer_stats.live).unwrap_or(0),
                state: match stats.state {
                    TorrentStatsState::Paused => "paused".into(),
                    _ if stats.finished => "seeding".into(),
                    _ => "downloading".into(),
                },
                files,
            })
        })
        .collect()
}
```

#[tauri::command]
pub async fn set_global_speed_limit(
    down_bps: Option<u64>,
    up_bps: Option<u64>,
    state: State<'_, Arc<AppState>>,
) -> Result<(), String> {
    use std::num::NonZeroU32;

    if let Some(bps) = down_bps {
        if let Some(nz) = NonZeroU32::new(bps as u32) {
            state.session.ratelimits.set_download_bps(Some(nz));
        }
    }
    if let Some(bps) = up_bps {
        if let Some(nz) = NonZeroU32::new(bps as u32) {
            state.session.ratelimits.set_upload_bps(Some(nz));
        }
    }
    *state.speed_limit_down.write() = down_bps;
    *state.speed_limit_up.write() = up_bps;
    Ok(())
}

pub async fn pause_all_downloads(state: State<'_, Arc<AppState>>) -> Result<(), String> {
    let info_hashes: Vec<String> = state.monitored_torrents.read().iter().cloned().collect();
    for info_hash in info_hashes {

```kotlin
// src-tauri/gen/android/app/src/main/java/com/sternelee/torplay/MediaStoreHelper.kt
package com.sternelee.torplay

import android.content.ContentValues
import android.content.Context
import android.net.Uri
import android.os.Build
import android.provider.MediaStore
import java.io.File
import java.io.FileInputStream

object MediaStoreHelper {

    fun insertDownload(context: Context, localPath: String, filename: String): String {
        val contentValues = ContentValues().apply {
            put(MediaStore.Downloads.DISPLAY_NAME, filename)
            put(MediaStore.Downloads.MIME_TYPE, getMimeType(filename))
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                put(MediaStore.Downloads.RELATIVE_PATH, "Download/TorPlay")
            }
        }

        val resolver = context.contentResolver
        val uri: Uri? = resolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, contentValues)

        if (uri != null) {
            val sourceFile = File(localPath)
            if (sourceFile.exists()) {
                resolver.openOutputStream(uri)?.use { outputStream ->
                    FileInputStream(sourceFile).use { inputStream ->
                        inputStream.copyTo(outputStream)
                    }
                }
                return uri.toString()
            }
        }
        return ""
    }

    private fun getMimeType(filename: String): String {
        return when {
            filename.endsWith(".mp4") || filename.endsWith(".m4v") -> "video/mp4"
            filename.endsWith(".mkv") -> "video/x-matroska"
            filename.endsWith(".webm") -> "video/webm"
            filename.endsWith(".mov") -> "video/quicktime"
            filename.endsWith(".avi") -> "video/x-msvideo"
            filename.endsWith(".srt") -> "application/x-subtitle"
            filename.endsWith(".vtt") -> "text/vtt"
            else -> "application/octet-stream"
        }
    }
}
```

import { createSignal } from "solid-js";
import { invoke } from "@tauri-apps/api/core";

export interface FileInfo {
  index: number;
  name: string;
  sizeBytes: number;
  downloadedBytes: number;
  path: string | null;
  isVideo: boolean;
}

export interface DownloadTask {
  infoHash: string;
  name: string;
  progressPercent: number;
  downloadSpeedBps: number;
  uploadSpeedBps: number;
  peersConnected: number;
  state: "downloading" | "paused" | "seeding" | "complete" | "error";
  files: FileInfo[];
}

export const [downloads, setDownloads] = createSignal<DownloadTask[]>([]);
export const [downSpeedLimit, setDownSpeedLimit] = createSignal<number | null>(null);
export const [upSpeedLimit, setUpSpeedLimit] = createSignal<number | null>(null);

export async function refreshDownloads() {
  try {
    const tasks = await invoke<DownloadTask[]>("get_active_downloads");
    setDownloads(tasks);
  } catch (e) {
    console.error("Failed to refresh downloads:", e);
  }
}

export async function setSpeedLimit(downKbps: number | null, upKbps: number | null) {
  await invoke("set_global_speed_limit", {
    downBps: downKbps ? downKbps * 1024 : null,
    upBps: upKbps ? upKbps * 1024 : null,
  });
  setDownSpeedLimit(downKbps);
  setUpSpeedLimit(upKbps);
}

export async function pauseAll() {
  await invoke("pause_all_downloads");
  await refreshDownloads();
}

export async function resumeAll() {
  await invoke("resume_all_downloads");
  await refreshDownloads();
}

export async function openDownloadedFile(path: string) {
  await invoke("open_in_file_manager", { path });
}

export async function exportToMediaStore(infoHash: string, fileIndex: number) {
  return invoke<string>("export_file", { infoHash, fileIndex });
}

```rust
#[tauri::command]
pub async fn open_in_file_manager(path: String) -> Result<(), String> {
    let path = PathBuf::from(&path);
    if !path.exists() {
        return Err("File not found".to_string());
    }

    #[cfg(target_os = "macos")]
    {
        std::process::Command::new("open")
            .args(["-R", &path.to_string_lossy()])
            .spawn()
            .map_err(|e| e.to_string())?;
    }

    #[cfg(target_os = "windows")]
    {
        std::process::Command::new("explorer")
            .args(["/select,", &path.to_string_lossy()])
            .spawn()
            .map_err(|e| e.to_string())?;
    }

    #[cfg(target_os = "linux")]
    {
        std::process::Command::new("xdg-open")
            .arg(path.parent().unwrap_or(&path))
            .spawn()
            .map_err(|e| e.to_string())?;
    }

    Ok(())
}
```
